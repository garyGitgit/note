import heapq
h = []
INF = float("inf")
VISITED = 1


def dijkstra(weight, start):
    visited = [0]*6
    min_cost = [INF]*6

    # initialization
    heapq.heappush(h, (min_cost[start], start))

    # check until queue is empty
    while h:
        _, v = heapq.heappop(h)
        print ("pop v: ", v)
        visited[v] = VISITED
        for i in range(6):
            if v == 3 and i == 4:
                print ("i: {}")
            if not visited[i]:
                if weight[start][v] + weight[v][i] < min_cost[i]:
                    min_cost[i] = weight[start][v] + weight[v][i]
                    heapq.heappush(h, (min_cost[i], i))
            else:
                # pass visited node or unreachable node
                pass

    # print all shortest weight for all nodes
    print ("min cost: ", min_cost)

def main():
    weight = [
        [0,   2,   5,   1,   INF, INF],
        [2,   0,   3,   3,   INF, INF],
        [5,   3,   0,   3,   1,   5],
        [1,   2,   3,   0,   1,   INF],
        [INF, INF, 5,   1,   0,   2],
        [INF, INF, 5,   INF, 2,   0],
    ]

    dijkstra(weight, 0)

if __name__ == "__main__":
    main()
