UP = 1
DOWN = 0
class MinHeap:

    def __init__(self):
        self.heap = [0] # dummy. heap starts from 1

    def __compare(self, c1, c2):
        return True if self.heap[c1] < self.heap[c2] else False

    def __heapify(self, start, d):
        # from top to bottom when pop

        assert d == 0 or d == 1
        cursor = start
        
        if d == DOWN:
            while 2*cursor <= len(self.heap)-1:
                lc, rc = 2*cursor, 2*cursor + 1
                if rc <= len(self.heap)-1:
                    child = lc if self.__compare(lc, rc) else rc
                else:
                    child = lc
                # swap
                print ("cursor: {}, child: {}".format(cursor, child))
                print(self.heap[1:])
                if not self.__compare(cursor, child):
                    self.heap[cursor], self.heap[child] = self.heap[child], self.heap[cursor]
                    print(self.heap[1:])
                cursor = child
        # from bottom to top when insert
        else:
            p = int(cursor/2)
            while p >= 1:
                if self.__compare(cursor, p):
                    self.heap[cursor], self.heap[p] = self.heap[p], self.heap[cursor]
                    cursor = p
                    # update parent
                    p = int(cursor/2)
                else:
                    break

    def push(self, key):
        self.heap.append(key)
        print ("push {}".format(key))
        self.__heapify(len(self.heap)-1, UP)

    def pop(self):
        popkey = self.heap[1] if len(self.heap)-1 > 0 else -1
        self.heap[1] = self.heap[len(self.heap)-1]
        del self.heap[len(self.heap)-1]
        self.__heapify(1, DOWN)
        return popkey
    
def main():
    h = MinHeap()
    h.push(1)
    h.push(2)
    h.push(3)
    h.push(4)
    print ("after insert: ", h.heap[1:])
    while len(h.heap)-1 > 0:
        print(h.pop(), end=" ")
    print()

if __name__ == "__main__":
    main()
